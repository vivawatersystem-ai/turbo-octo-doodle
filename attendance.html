// ==== الإعدادات ==== var COMPANY_LAT = 29.991159; var COMPANY_LNG = 31.152182; var MAX_DISTANCE_METERS = 150; // نطاق مسموح حوالين الشركة (متر) var DEVICES_SHEET_NAME = "الأجهزة المسجلة"; var EMPLOYEES = { "EMP01": "يوسف حمدي", "EMP02": "عمرو محمد", "EMP03": "محمد الطباخ", "EMP04": "خالد الطباخ", "EMP05": "سندس", "EMP06": "رودينا", "EMP07": "فاطمة", "EMP08": "جنة", "EMP09": "احمد سلام", "EMP10": "حسام كمال", "EMP11": "عبدالسلام", "EMP12": "احمد مجدي" }; function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; // شيت الحضور الأساسي (أول شيت) var data = JSON.parse(e.postData.contents); var code = data.code; var type = data.type; // "حضور" او "انصراف" var lat = parseFloat(data.lat); var lng = parseFloat(data.lng); var deviceId = data.deviceId || ""; // تحقق من الكود if (!EMPLOYEES[code]) { return jsonResponse("error", "كود الموظف غير صحيح"); } var name = EMPLOYEES[code]; // ==== تحقق من ربط الجهاز ==== var devicesSheet = getOrCreateDevicesSheet(ss); var deviceCheck = checkDevice(devicesSheet, code, name, deviceId); if (!deviceCheck.allowed) { return jsonResponse("error", deviceCheck.message); } // حساب المسافة من الشركة var distance = getDistanceMeters(lat, lng, COMPANY_LAT, COMPANY_LNG); var insideRange = distance <= MAX_DISTANCE_METERS; var now = new Date(); var dateStr = Utilities.formatDate(now, "GMT+2", "yyyy-MM-dd"); var timeStr = Utilities.formatDate(now, "GMT+2", "HH:mm:ss"); sheet.appendRow([ now, name, code, type, dateStr, timeStr, lat, lng, insideRange ? "نعم" : "لا" ]); if (!insideRange) { return jsonResponse("warning", "تم التسجيل، لكنك خارج نطاق الشركة (" + Math.round(distance) + " متر)"); } return jsonResponse("success", "تم تسجيل " + type + " بنجاح لـ " + name); } function checkDevice(devicesSheet, code, name, deviceId) { if (!deviceId) { return { allowed: false, message: "تعذر التعرف على جهازك، جرب تفتح الصفحة من جديد." }; } var values = devicesSheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (values[i][0] === code) { var savedDeviceId = values[i][2]; if (savedDeviceId === deviceId) { return { allowed: true }; // نفس الجهاز المسجل } else { return { allowed: false, message: "هذا الكود مسجل بالفعل على جهاز آخر. تواصل مع الإدارة لو ده جهازك الجديد." }; } } } // مفيش تسجيل قبل كده لهذا الكود -> سجّل الجهاز ده كأول جهاز devicesSheet.appendRow([code, name, deviceId, new Date()]); return { allowed: true }; } function getOrCreateDevicesSheet(ss) { var sheet = ss.getSheetByName(DEVICES_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(DEVICES_SHEET_NAME); sheet.appendRow(["كود الموظف", "اسم الموظف", "معرف الجهاز", "تاريخ أول تسجيل"]); } return sheet; } function jsonResponse(status, message) { return ContentService.createTextOutput(JSON.stringify({ status: status, message: message })).setMimeType(ContentService.MimeType.JSON); } function getDistanceMeters(lat1, lng1, lat2, lng2) { var R = 6371000; var dLat = (lat2 - lat1) * Math.PI / 180; var dLng = (lng2 - lng1) * Math.PI / 180; var a = Math.sin(dLat/2) * Math.sin(dLat/2) + Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * Math.sin(dLng/2) * Math.sin(dLng/2); var c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)); return R * c; }
