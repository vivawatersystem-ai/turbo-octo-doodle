// ==== الإعدادات ==== var COMPANY_LAT = 29.991159; var COMPANY_LNG = 31.152182; var MAX_DISTANCE_METERS = 150; // نطاق مسموح حوالين الشركة (متر) var EMPLOYEES = { "EMP01": "يوسف حمدي", "EMP02": "عمرو محمد", "EMP03": "محمد الطباخ", "EMP04": "خالد الطباخ", "EMP05": "سندس", "EMP06": "رودينا", "EMP07": "فاطمة", "EMP08": "جنة", "EMP09": "احمد سلام", "EMP10": "حسام كمال", "EMP11": "عبدالسلام", "EMP12": "احمد مجدي" }; function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); var code = data.code; var type = data.type; // "حضور" او "انصراف" var lat = parseFloat(data.lat); var lng = parseFloat(data.lng); // تحقق من الكود if (!EMPLOYEES[code]) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "كود الموظف غير صحيح" })).setMimeType(ContentService.MimeType.JSON); } var name = EMPLOYEES[code]; // حساب المسافة من الشركة var distance = getDistanceMeters(lat, lng, COMPANY_LAT, COMPANY_LNG); var insideRange = distance <= MAX_DISTANCE_METERS; var now = new Date(); var dateStr = Utilities.formatDate(now, "GMT+2", "yyyy-MM-dd"); var timeStr = Utilities.formatDate(now, "GMT+2", "HH:mm:ss"); sheet.appendRow([ now, name, code, type, dateStr, timeStr, lat, lng, insideRange ? "نعم" : "لا" ]); if (!insideRange) { return ContentService.createTextOutput(JSON.stringify({ status: "warning", message: "تم التسجيل، لكنك خارج نطاق الشركة (" + Math.round(distance) + " متر)" })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "تم تسجيل " + type + " بنجاح لـ " + name })).setMimeType(ContentService.MimeType.JSON); } function getDistanceMeters(lat1, lng1, lat2, lng2) { var R = 6371000; var dLat = (lat2 - lat1) * Math.PI / 180; var dLng = (lng2 - lng1) * Math.PI / 180; var a = Math.sin(dLat/2) * Math.sin(dLat/2) + Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * Math.sin(dLng/2) * Math.sin(dLng/2); var c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)); return R * c; }
